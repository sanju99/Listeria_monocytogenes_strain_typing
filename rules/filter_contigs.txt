process FILTER_CONTIGS {

    tag "${sample} (${mode})"

    input:
    tuple val(sample), val(mode), path(fasta), path(depth_file)

    output:
    tuple val(sample), val(mode), path("filtered_contigs.fasta")

    script:
    """
    filter_contigs.py \
        --fasta ${contigs} \
        --depth ${depth_file} \
        --output filtered_contigs.fasta \
        --min_contig_length ${params.min_contig_length} \
        --min_median_coverage ${params.min_median_coverage}
    """
}


process QUAST_ON_FILTERED_CONTIGS {

    tag "${sample} (${mode}) (${kmer_set})"

    input:
    tuple val(sample), 
          val(mode), 
          val(preprocessing), 
          val(kmer_set), 
          path(contigs) 

    output:
    tuple val(sample), 
          val(mode), 
          val(preprocessing), 
          val(kmer_set), 
          path("quast/report.tsv")

    script:
    """
    quast.py ${contigs} ${scaffolds} -l filtered_contigs -o quast --threads ${task.cpus}
    """

}
