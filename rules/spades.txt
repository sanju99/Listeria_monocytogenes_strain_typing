process SPADES_ON_FASTP {

    tag "${sample} (${mode})"

    publishDir { "${params.outdir}/${sample}/fastp/spades/${mode}" }, mode: 'copy'

    input:
    tuple val(sample), val(mode), path(r1), path(r2), path(merged)

    output:
    tuple val(sample),
          val('fastp'),
          val(mode),
          path("kmers_through_77"),
          path("kmers_through_127")

    script:
    """
    # create the output directories
    mkdir -p kmers_through_77 kmers_through_127

    # force the k-mers because the breakpoint for using 77-mers or not is 150 bp, and the NextSeq 500 reads are 149 bp
    # but we need to use the same params as for the NextSeq 1000, which output 151 bp reads
    spades.py \
        --${mode} \
        -k 21,33,55,77 \
        -1 "${r1}" \
        -2 "${r2}" \
        --merged "${merged}" \
        -o kmers_through_127/spades_out \
        --threads ${task.cpus}

    # Save the K77 assembly before restarting
    cp kmers_through_127/spades_out/contigs.fasta kmers_through_77/contigs.fasta
    cp kmers_through_127/spades_out/scaffolds.fasta kmers_through_77/scaffolds.fasta

    spades.py \
        --restart-from k77 \
        -k 21,33,55,77,99,127 \
        -o kmers_through_127/spades_out \
        --threads ${task.cpus}
    """
}



process SPADES_ON_TRIMMOMATIC {

    tag "${sample} (${mode})"
    publishDir { "${params.outdir}/${sample}/trimmomatic_moap/spades/${mode}/kmers_through_77" }, mode: 'copy'

    input:
    tuple val(sample),
          val(mode),
          path("${sample}_R1.trimmed.fastq.gz"),
          path("${sample}_R2.trimmed.fastq.gz")

    output:
    tuple val(sample),
          val("trimmomatic_moap"),
          val(mode),
          path("spades_out")

    script:
    """
    # force the k-mers because the breakpoint for using 77-mers or not is 150 bp, and the NextSeq 500 reads are 149 bp
    # but we need to use the same params as for the NextSeq 1000, which output 151 bp
    spades.py \
        --${mode} \
        -k 21,33,55,77 \
        -1 "${sample}_R1.trimmed.fastq.gz" \
        -2 "${sample}_R2.trimmed.fastq.gz" \
        -o spades_out \
        --threads ${task.cpus}
    """
}
